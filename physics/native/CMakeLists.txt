cmake_minimum_required(VERSION 3.22)
project(mechanica_physics LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${BOX2D_SOURCE}" box2d EXCLUDE_FROM_ALL)
add_library(mechanica_physics STATIC physics_bridge.c)
target_link_libraries(mechanica_physics PUBLIC box2d)
find_package(JNI QUIET)
if(JNI_FOUND)
    add_library(mechanica_physics_jni SHARED physics_jni.c)
    target_include_directories(mechanica_physics_jni PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(mechanica_physics_jni PRIVATE mechanica_physics)
endif()
